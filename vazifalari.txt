from sqlalchemy import create_engine
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import sessionmaker
from app.config import settings

# Bazaga ulanish
engine = create_engine(settings.DATABASE_URL)

# Session yaratish (har bir so'rov uchun)
SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine)

# Barcha modellar shu Base dan meros oladi
Base = declarative_base()

# Har bir API so'rovda yangi session ochib, oxirida yopish
def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()

TUSHUNTIRISH:
- engine — bazaga ulanish "dvigateli"
- SessionLocal — har bir so'rov uchun alohida session (ulanish) yaratadi
- Base — barcha modellar (jadvallar) shu klassdan meros oladi

from sqlalchemy import Column, Integer, String, DateTime, func
from app.database import Base

class User(Base):
    __tablename__ = "users"
    
    id = Column(Integer, primary_key=True, index=True)
    email = Column(String(255), unique=True, index=True, nullable=False)
    phone = Column(String(20), nullable=True)
    password_hash = Column(String(255), nullable=False)
    full_name = Column(String(255), nullable=False)
    role = Column(String(20), default="student")  # "student" yoki "admin"
    created_at = Column(DateTime, server_default=func.now())

TUSHUNTIRISH:
- __tablename__ — bazadagi jadval nomi
- Column — jadval ustuni (column)
- primary_key=True — asosiy kalit (har bir qator uchun yagona raqam)
- unique=True — takrorlanmasligi kerak (masalan, email)
- nullable=False — bo'sh bo'lishi MUMKIN EMAS
- nullable=True — bo'sh bo'lishi MUMKIN
- default="student" — agar ko'rsatilmasa, "student" bo'ladi
- server_default=func.now() — avtomatik hozirgi vaqtni yozadi

def check_answer(user_answer: str, correct_answer: str) -> bool:
    """Javobni tekshirish — katta-kichik harfga e'tibor bermaydi"""
    return user_answer.strip().lower() == correct_answer.strip().lower()


def calculate_reading_score(correct_count: int, total_questions: int = 40) -> float:
    """
    IELTS Reading ball hisoblash.
    40 ta savoldan nechta to'g'ri — band score ga o'tkazish.
    """
    # IELTS Academic Reading band score jadvali (taxminiy)
    if correct_count >= 39: return 9.0
    if correct_count >= 37: return 8.5
    if correct_count >= 35: return 8.0
    if correct_count >= 33: return 7.5
    if correct_count >= 30: return 7.0
    if correct_count >= 27: return 6.5
    if correct_count >= 23: return 6.0
    if correct_count >= 19: return 5.5
    if correct_count >= 15: return 5.0
    if correct_count >= 13: return 4.5
    if correct_count >= 10: return 4.0
    if correct_count >= 6:  return 3.5
    if correct_count >= 4:  return 3.0
    return 2.5


def calculate_listening_score(correct_count: int, total_questions: int = 40) -> float:
    """
    IELTS Listening ball hisoblash.
    """
    if correct_count >= 39: return 9.0
    if correct_count >= 37: return 8.5
    if correct_count >= 35: return 8.0
    if correct_count >= 32: return 7.5
    if correct_count >= 30: return 7.0
    if correct_count >= 26: return 6.5
    if correct_count >= 23: return 6.0
    if correct_count >= 18: return 5.5
    if correct_count >= 16: return 5.0
    if correct_count >= 13: return 4.5
    if correct_count >= 10: return 4.0
    if correct_count >= 6:  return 3.5
    if correct_count >= 4:  return 3.0
    return 2.5


def calculate_overall_band(reading: float, listening: float,
                           writing: float, speaking: float) -> float:
    """
    Umumiy band score hisoblash.
    4 ta bo'lim o'rtachasi, 0.5 ga yaxlitlanadi.
    """
    average = (reading + listening + writing + speaking) / 4
    # 0.5 ga yaxlitlash (IELTS qoidasi)
    return round(average * 2) / 2
